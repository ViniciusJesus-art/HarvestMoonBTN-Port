foreach(layout RANGE 0 2)
    string(REGEX MATCH "static u16 title_layout${layout}_cells\\[1126\\] = \\{[^}]+\\}" layout_table "${title_layout_source}")
    if(NOT layout_table)
        message(FATAL_ERROR "Cannot find original initial field layout ${layout}")

    string(APPEND field_layout_source "${layout_table};\n")
