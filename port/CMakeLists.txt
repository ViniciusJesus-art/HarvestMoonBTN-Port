# The native host never links the PSY-Q SDK or MIPS game binaries.
include(cmake/sdl3.cmake)
add_library(hmbtn_port_core STATIC core/psx_data.c graphics/asset_decode.c graphics/window_frame.c game/title_flow.c game/new_game.c audio/psx_audio.c)
target_include_directories(hmbtn_port_core PUBLIC core graphics game input audio)
if(NOT MSVC)
    target_link_libraries(hmbtn_port_core PUBLIC m)
endif()
target_compile_features(hmbtn_port_core PUBLIC c_std_11)
set_target_properties(hmbtn_port_core PROPERTIES C_EXTENSIONS OFF)

add_library(hmbtn_host STATIC core/app.c platform/sdl_host.c
    graphics/diagnostic_screen.c graphics/title_preview.c input/window_input.c
    input/game_input.c filesystem/resource_file.c filesystem/data_directory.c audio/title_audio.c graphics/new_game_ui.c)
target_include_directories(hmbtn_host PUBLIC core platform graphics input filesystem game)
target_link_libraries(hmbtn_host PUBLIC hmbtn_port_core SDL3::SDL3)
set_target_properties(hmbtn_host PROPERTIES C_EXTENSIONS OFF)

add_executable(hmbtn_native core/main.c)
target_link_libraries(hmbtn_native PRIVATE hmbtn_host)
set_target_properties(hmbtn_native PROPERTIES C_EXTENSIONS OFF)
if(MINGW)
    target_link_options(hmbtn_native PRIVATE -static)
endif()
# System SDL3 may be shared. Keep its DLL beside the executable on Windows.
get_target_property(sdl3_type SDL3::SDL3 TYPE)
if(WIN32 AND sdl3_type STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET hmbtn_native POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:SDL3::SDL3>"
            "$<TARGET_FILE_DIR:hmbtn_native>" VERBATIM)
endif()
install(TARGETS hmbtn_native RUNTIME DESTINATION .)
if(DEFINED SDL3_SOURCE_DIR AND EXISTS "${SDL3_SOURCE_DIR}/LICENSE.txt")
    install(FILES "${SDL3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME SDL3.txt)
endif()
if(WIN32 AND sdl3_type STREQUAL "SHARED_LIBRARY")
    install(FILES "$<TARGET_FILE:SDL3::SDL3>" DESTINATION .)
endif()
if(WIN32)
    install(FILES "${CMAKE_SOURCE_DIR}/docs/port/WINDOWS_PACKAGE.txt" DESTINATION . RENAME README.txt)
endif()

foreach(target hmbtn_port_core hmbtn_host hmbtn_native)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
