# The native host never links the PSY-Q SDK or MIPS game binaries.
include(cmake/sdl3.cmake)
# Reuse the existing title's three initial field layouts without linking PS1
# code or duplicating its map data in the native source tree.
set(title_source "${CMAKE_SOURCE_DIR}/src/overlay/27_GF_TITLE/title.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${title_source}")
file(READ "${title_source}" title_layout_source)
set(field_layout_source "/* Generated from title_fill_field_layout's existing tables. */\n")
foreach(layout RANGE 0 2)
    string(REGEX MATCH "static u16 title_layout${layout}_cells\\[1126\\] = \\{[^}]+\\}" layout_table "${title_layout_source}")
    if(NOT layout_table)
        message(FATAL_ERROR "Cannot find original initial field layout ${layout}")
    endif()
    string(REPLACE "static u16" "static const uint16_t" layout_table "${layout_table}")
    string(APPEND field_layout_source "${layout_table};\n")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/field_layouts.inc" "${field_layout_source}")
set(text_bank_source "${CMAKE_SOURCE_DIR}/src/main/text/text_bank.c")
set(text_source "${CMAKE_SOURCE_DIR}/src/main/text/text.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${text_bank_source}" "${text_source}")
file(READ "${text_bank_source}" house_text_source)
set(house_labels "/* Generated from the original C interface labels. */\n")
foreach(label year storage_equip storage_toolbox slash sleep_menu
        how_to_operate_configuration how_to_operate_buttons configuration_screen
        type_1 type_2 type_3 on off stereo mono yes no)
    string(REGEX MATCH "\\.${label} = \\{[^}]+\\}" house_label "${house_text_source}")
    if(NOT house_label)
        message(FATAL_ERROR "Missing original label ${label}")
    endif()
    string(REPLACE ".${label} =" "static const uint16_t house_${label}[] =" house_label "${house_label}")
    string(APPEND house_labels "${house_label};\n")
endforeach()
file(READ "${text_source}" house_text_source)
foreach(label season_text_codes weekday_text_codes)
    string(REGEX MATCH "static u16 ${label}\\[[0-9]+\\] = \\{[^}]+\\}" house_label "${house_text_source}")
    string(REPLACE "static u16" "static const uint16_t" house_label "${house_label}")
    string(APPEND house_labels "${house_label};\n")
endforeach()
string(REPLACE "TEXT_CODE_SPACE" "0x0FFD" house_labels "${house_labels}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/house_labels.inc" "${house_labels}")
add_library(hmbtn_port_core STATIC core/psx_data.c graphics/asset_decode.c graphics/window_frame.c
    graphics/model_decode.c graphics/model_render.c graphics/scene_decode.c game/title_flow.c game/new_game.c
    game/event_script.c game/farm_intro.c game/ground.c game/house_control.c game/house_interaction.c
    game/house_sleep.c audio/psx_audio.c audio/xa_decode.c)
target_include_directories(hmbtn_port_core PUBLIC core graphics game input audio)
if(NOT MSVC)
    target_link_libraries(hmbtn_port_core PUBLIC m)
endif()
target_compile_features(hmbtn_port_core PUBLIC c_std_11)
set_target_properties(hmbtn_port_core PROPERTIES C_EXTENSIONS OFF)

find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/make_icon.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/packaging/make_icon.py"
    "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
include(cmake/import_table.cmake)
add_library(hmbtn_host STATIC core/app.c platform/sdl_host.c platform/app_icon.c
    installer/disc_import.c installer/sha256.c installer/setup.c
    graphics/diagnostic_screen.c graphics/title_preview.c input/window_input.c
    input/game_input.c filesystem/resource_file.c filesystem/data_directory.c audio/title_audio.c audio/xa_stream.c
    graphics/new_game_ui.c graphics/house_ui.c graphics/scene_preview.c)
target_include_directories(hmbtn_host PUBLIC core platform graphics input filesystem game installer)
target_include_directories(hmbtn_host PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(hmbtn_host PUBLIC hmbtn_port_core SDL3::SDL3)
set_target_properties(hmbtn_host PROPERTIES C_EXTENSIONS OFF)

add_executable(hmbtn_native core/main.c)
target_link_libraries(hmbtn_native PRIVATE hmbtn_host)
set_target_properties(hmbtn_native PROPERTIES C_EXTENSIONS OFF)
if(MINGW)
    target_link_options(hmbtn_native PRIVATE -static)
endif()
# System SDL3 may be shared. Keep its DLL beside the executable on Windows.
get_target_property(sdl3_type SDL3::SDL3 TYPE)
if(WIN32 AND sdl3_type STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET hmbtn_native POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:SDL3::SDL3>"
            "$<TARGET_FILE_DIR:hmbtn_native>" VERBATIM)
endif()
include(packaging/packaging.cmake)

foreach(target hmbtn_port_core hmbtn_host hmbtn_native)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
